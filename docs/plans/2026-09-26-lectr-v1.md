**Goal:** Build `lectr`, a CLI that converts DRM-free EPUB/PDF/MOBI/AZW3/TXT/Markdown books into chaptered M4B (or per-chapter MP3) audiobooks with the local Kokoro-82M TTS model, installable with uv and Homebrew.

**Architecture:** Three stages: `readers/` turns a file into a `Book` of `Chapter`s, `pipeline.py` synthesizes each chapter to a cached WAV (resume on rerun), and `audio.py` assembles WAVs with ffmpeg. Settings resolve as CLI flag > YAML file > default, and `lectr setup` writes the YAML. Everything except `tts.KokoroSynthesizer` is tested without the model.

**Tech Stack:** Python 3.13, uv (`uv_build` backend), kokoro-onnx 0.6.x (ONNX Runtime, bundled espeak-ng), mobi 0.4.1, pypdf[crypto] 6.x, PyYAML 6, platformdirs 4, ffmpeg (external), pytest, ruff, mypy, GitHub Actions.

**Spec:** `docs/specs/2026-09-26-lectr-design.md` (read §12 "Planning verification results" — it amends earlier sections).

## Review Focus

1. EPUBs that use IDPF font obfuscation list only font files in `META-INF/encryption.xml` and are DRM-free — they must convert, not be rejected (test in Task 4).
2. EPUBs where one chapter spans several spine files (common after Calibre conversion) must yield one chapter, not extra "Chapter N" entries (test in Task 4).
3. Owner-password (permissions-only) PDFs are DRM-free for reading and must open (test in Task 5).
4. Plain-text books saved as Windows-1252 must read instead of crashing on decode (test in Task 3).
5. `--overwrite` after a previous run that produced more M4B parts must not leave stale `Part N of M` files next to the new output (test in Task 12).

Also pinned: cache paths containing an apostrophe survive ffmpeg's concat list (Task 11); books with no chapter markers (a PDF without bookmarks, a TXT without headings) are split into ~5,000-word segments so memory stays bounded and resume stays useful (Tasks 2–3, [G1]).

`src/lectr/readers/text.py`:

```python
"""Text cleanup shared by every reader, and HTML-to-text for EPUB/MOBI content."""

import math
import re
from html.parser import HTMLParser
from typing import NamedTuple

from lectr.book import Chapter

_HYPHEN_BREAK = re.compile(r"(\w)-\n(\w)")
_PARAGRAPH_BREAK = re.compile(r"\n[ \t]*\n")

_SKIP = frozenset({"script", "style", "head", "title"})
_BLOCK = frozenset(
    {
        "p",
        "div",
        "br",
        "hr",
        "li",
        "tr",
        "pre",
        "blockquote",
        "section",
        "article",
        "h1",
        "h2",
        "h3",
        "h4",
        "h5",
        "h6",
        "mbp:pagebreak",
    }
)
_HEADINGS = frozenset({"h1", "h2", "h3"})
# Superscripts made only of digits/symbols are footnote markers, not words.
_NOTE_MARK = re.compile(r"[\d\s,*†‡§\[\]]+")


def clean_text(text: str) -> str:
    """Normalize whitespace into paragraphs separated by one blank line."""
    text = text.replace("\r\n", "\n").replace("\r", "\n")
    text = _HYPHEN_BREAK.sub(r"\1\2", text)
    paragraphs = (" ".join(p.split()) for p in _PARAGRAPH_BREAK.split(text))
    return "\n\n".join(p for p in paragraphs if p)


class HtmlText(NamedTuple):
    text: str
    first_heading: str | None


class _Extractor(HTMLParser):
    def __init__(self) -> None:
        super().__init__(convert_charrefs=True)
        self.parts: list[str] = []
        self.first_heading: str | None = None
        self._skip_tag: str | None = None
        self._skip_depth = 0
        self._sup: list[str] | None = None
        self._heading: list[str] | None = None

    def handle_starttag(self, tag: str, attrs: list[tuple[str, str | None]]) -> None:
        if self._skip_tag is not None:
            if tag == self._skip_tag:
                self._skip_depth += 1
            return
        epub_type = dict(attrs).get("epub:type") or ""
        if tag in _SKIP or "noteref" in epub_type.split():
            self._skip_tag, self._skip_depth = tag, 1
            return
        if tag in _BLOCK:
            self.parts.append("\n\n")
        if tag == "sup":
            self._sup = []
        if tag in _HEADINGS and self.first_heading is None:
            self._heading = []

    def handle_endtag(self, tag: str) -> None:
        if self._skip_tag is not None:
            if tag == self._skip_tag:
                self._skip_depth -= 1
                if self._skip_depth == 0:
                    self._skip_tag = None
            return
        if tag == "sup" and self._sup is not None:
            text = "".join(self._sup)
            self._sup = None
            if not _NOTE_MARK.fullmatch(text):
                self.parts.append(text)
        if tag in _HEADINGS and self._heading is not None:
            heading = " ".join("".join(self._heading).split())
            self._heading = None
            if heading:
                self.first_heading = heading
        if tag in _BLOCK:
            self.parts.append("\n\n")

    def handle_data(self, data: str) -> None:
        if self._skip_tag is not None:
            return
        if self._heading is not None:
            self._heading.append(data)
        if self._sup is not None:
            self._sup.append(data)
        else:
            self.parts.append(data)


def html_to_text(html: str) -> HtmlText:
    parser = _Extractor()
    parser.feed(html)
    parser.close()
    return HtmlText(clean_text("".join(parser.parts)), parser.first_heading)


# [G1] kokoro-onnx returns a whole chapter as one in-memory array and the stdlib `wave`
# format tops out at 4 GiB (~24.8 h at 24 kHz), so no chapter is synthesized in one piece
# above ~5,000 words (roughly half an hour). Long chapters become equal-sized parts.
MAX_SEGMENT_WORDS = 5_000
_SENTENCE_END = re.compile(r"(?<=[.!?])\s+")


def split_long_chapter(chapter: Chapter, max_words: int = MAX_SEGMENT_WORDS) -> list[Chapter]:
    total = len(chapter.text.split())
    if total <= max_words:
        return [chapter]
    grain = max(1, max_words // 50)  # pieces this small let the parts come out even
    count = math.ceil(total / (max_words - grain))  # headroom: no part passes max_words
    target = total / count
    parts: list[list[list[str]]] = [[] for _ in range(count)]  # part -> paragraphs -> pieces
    position = 0
    for paragraph in chapter.text.split("\n\n"):
        started: set[int] = set()
        for sentence in _SENTENCE_END.split(paragraph):
            words = sentence.split()
            for start in range(0, len(words), grain):
                piece = words[start : start + grain]
                # A piece belongs to the part its midpoint falls in.
                index = min(count - 1, int((position + len(piece) / 2) // target))
                if index not in started:
                    parts[index].append([])
                    started.add(index)
                parts[index][-1].append(" ".join(piece))
                position += len(piece)
    return [
        Chapter(
            f"{chapter.title} (Part {number} of {count})",
            "\n\n".join(" ".join(pieces) for pieces in paragraphs),
        )
        for number, paragraphs in enumerate(parts, 1)
    ]
```

```python
"""Plain text and Markdown readers."""

import re
from pathlib import Path

from lectr.book import Book, Chapter

_NUMBER = (
    r"(?:\d+|[ivxlcdm]+|one|two|three|four|five|six|seven|eight|nine|ten|eleven|twelve"
    r"|thirteen|fourteen|fifteen|sixteen|seventeen|eighteen|nineteen|twenty|thirty|forty"
    r"|fifty|sixty|seventy|eighty|ninety|hundred)"
    r"(?:[-\s](?:one|two|three|four|five|six|seven|eight|nine))?"
)
# A heading is a whole short line: "Chapter 12", "CHAPTER XII. Title", "Part One: Title".
# Prose such as "Chapter 3 was the best" is rejected because any text after the number
# must follow punctuation.
_DASHES = chr(0x2013) + chr(0x2014)  # en/em dash, spelled out for ruff RUF001 [G2]
_TXT_HEADING = re.compile(
    rf"^[ \t]*(?:chapter|part)[ \t]+{_NUMBER}\b[ \t]*(?:[.:{_DASHES}-][^\n]{{0,70}})?[ \t]*$",
    re.IGNORECASE | re.MULTILINE,
)
_GUTENBERG_START = re.compile(r"^\*\*\* ?START OF (?:THE|THIS) PROJECT GUTENBERG.*$", re.I | re.M)
_GUTENBERG_END = re.compile(r"^\*\*\* ?END OF (?:THE|THIS) PROJECT GUTENBERG", re.I | re.M)
_MD_FRONT_MATTER = re.compile(r"\A---[ \t]*\n.*?^(?:---|\.\.\.)[ \t]*$\n?", re.S | re.M)
_MD_SETEXT = re.compile(r"^([^\n#>\-*][^\n]*)\n(=+|-+)[ \t]*$", re.M)
_MD_TABLE_RULE = re.compile(
    r"^[ \t]*\|?[ \t]*:?-{3,}:?[ \t]*(?:\|[ \t]*:?-{3,}:?[ \t]*)*\|?[ \t]*\n", re.M
)
_MD_HEADING = re.compile(r"^(#{1,2})[ \t]+(.+?)[ \t]*#*[ \t]*$", re.MULTILINE)
_MD_FENCE = re.compile(r"^(```|~~~)[^\n]*\n.*?^\1[ \t]*$", re.MULTILINE | re.DOTALL)
_MD_RULES: tuple[tuple[re.Pattern[str], str], ...] = (
    (re.compile(r"!\[[^\]]*\]\([^)]*\)"), ""),  # images
    (re.compile(r"^[ \t]*(?:[-*_][ \t]*){3,}$", re.MULTILINE), ""),  # horizontal rules
    (re.compile(r"\[([^\]]+)\]\([^)]*\)"), r"\1"),  # links keep their text
    (re.compile(r"^#{1,6}[ \t]+", re.MULTILINE), ""),  # deeper headings
    (re.compile(r"^[ \t]*>[ \t]?", re.MULTILINE), ""),  # blockquotes
    (re.compile(r"^[ \t]*(?:[-*+]|\d+\.)[ \t]+", re.MULTILINE), ""),  # list markers
    (re.compile(r"<[^>]+>"), ""),  # inline HTML
    (re.compile(r"[ \t]*\|[ \t]*"), " "),  # table cell separators
    (re.compile(r"`([^`]*)`"), r"\1"),  # inline code
    (re.compile(r"(\*\*|__|\*|_)(\S(?:.*?\S)?)\1"), r"\2"),  # emphasis
)


def decode_text(data: bytes) -> str:
    """UTF-8 (with or without BOM), else Windows-1252, which never fails with 'replace'."""
    try:
        return data.decode("utf-8-sig")
    except UnicodeDecodeError:
        return data.decode("cp1252", errors="replace")


def read_text(path: Path) -> Book:
    text = _strip_gutenberg(decode_text(path.read_bytes()))
    headings = [
        (m.start(), m.end(), " ".join(m.group(0).split())) for m in _TXT_HEADING.finditer(text)
    ]
    return Book(path.stem, "", None, tuple(_split(text, headings, path.stem)))


def read_markdown(path: Path) -> Book:
    text = _MD_FRONT_MATTER.sub("", decode_text(path.read_bytes()))
    text = _MD_FENCE.sub("", text)
    # Setext headings ("Title" over ==== or ----) become ATX so one pattern finds them all.
    # This runs before table-rule removal, which would otherwise eat the ---- underline.
    text = _MD_SETEXT.sub(lambda m: ("# " if m.group(2)[0] == "=" else "## ") + m.group(1), text)
    text = _MD_TABLE_RULE.sub("", text)
    matches = list(_MD_HEADING.finditer(text))
    title = next((_strip_markdown(m.group(2)) for m in matches if m.group(1) == "#"), path.stem)
    headings = [(m.start(), m.end(), _strip_markdown(m.group(2))) for m in matches]
    chapters = [
        Chapter(chapter.title, _strip_markdown(chapter.text))
        for chapter in _split(text, headings, title)
    ]
    return Book(title, "", None, tuple(chapters))


def _strip_gutenberg(text: str) -> str:
    """Drop Project Gutenberg's header and licence, which are not part of the book."""
    start = _GUTENBERG_START.search(text)
    if start:
        text = text[start.end() :]
    end = _GUTENBERG_END.search(text)
    return text[: end.start()] if end else text


def _split(text: str, headings: list[tuple[int, int, str]], fallback_title: str) -> list[Chapter]:
    if not headings:
        return [Chapter(fallback_title, text)]
    chapters = [Chapter("Front Matter", text[: headings[0][0]])]
    for index, (_, end, title) in enumerate(headings):
        body_end = headings[index + 1][0] if index + 1 < len(headings) else len(text)
        chapters.append(Chapter(title, text[end:body_end]))
    return chapters


def _strip_markdown(text: str) -> str:
    for pattern, replacement in _MD_RULES:
        text = pattern.sub(replacement, text)
    return text
```

"""Turn a book file into a Book, whatever its format."""

from collections.abc import Callable
from pathlib import Path

from lectr.book import Book, Chapter
from lectr.errors import BookError
from lectr.readers.plain import read_markdown, read_text
from lectr.readers.text import clean_text, split_long_chapter

_READERS: dict[str, Callable[[Path], Book]] = {
    ".txt": read_text,
    ".md": read_markdown,
    ".markdown": read_markdown,
}
SUPPORTED_EXTENSIONS = tuple(sorted(_READERS))


def read_book(path: Path) -> Book:
    reader = _READERS.get(path.suffix.lower())
    if reader is None:
        raise BookError(
            f"Unsupported format '{path.suffix or path.name}'. "
            f"Supported: {', '.join(SUPPORTED_EXTENSIONS)}."
        )
    if not path.is_file():
        raise BookError(f"File not found: {path}")
    book = reader(path)
    chapters = tuple(
        segment
        for number, chapter in enumerate(book.chapters, 1)
        if (text := clean_text(chapter.text))
        for segment in split_long_chapter(
            Chapter(chapter.title.strip() or f"Chapter {number}", text)
        )
    )
    if not chapters:
        raise BookError(f"No readable text found in {path.name}.")
    return Book(
        " ".join(book.title.split()) or path.stem,
        " ".join(book.author.split()) or "Unknown",
        book.cover,
        chapters,
    )

from pathlib import Path


def _xhtml(title: str, body: str) -> str:
    return (
        '<?xml version="1.0" encoding="utf-8"?>\n'
        '<html xmlns="http://www.w3.org/1999/xhtml" xmlns:epub="http://www.idpf.org/2007/ops">'
        f"<head><title>{title}</title></head><body>{body}</body></html>"
    )


def make_epub(
    path: Path,
    *,
    title: str = "Test Book",
    author: str = "Ada Author",
    chapters: Sequence[tuple[str, str]] = (
        ("Chapter One", "<p>First words.</p>"),
        ("Chapter Two", "<p>Second words.</p>"),
    ),
    toc: str | None = "nav",
    cover: bytes | None = None,
    continuation: str | None = None,
    extra_files: dict[str, str | bytes] | None = None,
    toc_entries: Sequence[tuple[str, str]] | None = None,
    nav_in_spine: bool = False,
    guide_cover: bytes | None = None,
) -> Path:
    """Build a minimal EPUB 3.

    chapters: (title, body_html); each gets an <h1> and a TOC entry.
    toc: "nav" (EPUB 3 nav), "ncx" (EPUB 2 NCX), or None.
    continuation: body HTML of an untitled spine file placed right after chapter 1
        (simulates a chapter split across files).
    extra_files: additional or replacement archive members, e.g. META-INF/encryption.xml.
    toc_entries: explicit (href, title) TOC links, e.g. fragments like "text/ch1.xhtml#s2".
    nav_in_spine: also list the nav document in the reading order (some books do).
    guide_cover: cover image referenced only through an EPUB 2 <guide> cover page.
    """
    files: dict[str, str | bytes] = {}
    items: list[str] = []
    spine: list[str] = []
    entries: list[tuple[str, str]] = []
    for number, (chapter_title, body) in enumerate(chapters, 1):
        href = f"text/ch{number}.xhtml"
        files[f"OEBPS/{href}"] = _xhtml(chapter_title, f"<h1>{chapter_title}</h1>{body}")
        items.append(f'<item id="c{number}" href="{href}" media-type="application/xhtml+xml"/>')
        spine.append(f'<itemref idref="c{number}"/>')
        entries.append((href, chapter_title))
        if number == 1 and continuation is not None:
            files["OEBPS/text/ch1b.xhtml"] = _xhtml("", continuation)
            items.append(
                '<item id="c1b" href="text/ch1b.xhtml" media-type="application/xhtml+xml"/>'
            )
            spine.append('<itemref idref="c1b"/>')
    spine_attrs = ""
    guide = ""
    if toc_entries is not None:
        entries = list(toc_entries)
    if toc == "nav":
        links = "".join(f'<li><a href="{h}">{t}</a></li>' for h, t in entries)
        files["OEBPS/nav.xhtml"] = _xhtml(
            "Contents", f'<nav epub:type="toc"><ol>{links}</ol></nav>'
        )
        items.append(
            '<item id="nav" href="nav.xhtml" media-type="application/xhtml+xml" properties="nav"/>'
        )
        if nav_in_spine:
            spine.insert(0, '<itemref idref="nav"/>')
    elif toc == "ncx":
        points = "".join(
            f'<navPoint id="p{i}" playOrder="{i}"><navLabel><text>{t}</text></navLabel>'
            f'<content src="{h}"/></navPoint>'
            for i, (h, t) in enumerate(entries, 1)
        )
        files["OEBPS/toc.ncx"] = (
            '<?xml version="1.0" encoding="utf-8"?>'
            '<ncx xmlns="http://www.daisy.org/z3986/2005/ncx/" version="2005-1">'
            f"<navMap>{points}</navMap></ncx>"
        )
        items.append('<item id="ncx" href="toc.ncx" media-type="application/x-dtbncx+xml"/>')
        spine_attrs = ' toc="ncx"'
    if cover is not None:
        files["OEBPS/images/cover.jpg"] = cover
        items.append(
            '<item id="cover" href="images/cover.jpg" media-type="image/jpeg" '
            'properties="cover-image"/>'
        )
    if guide_cover is not None:
        files["OEBPS/images/front.jpg"] = guide_cover
        files["OEBPS/text/cover.xhtml"] = _xhtml("Cover", '<img src="../images/front.jpg"/>')
        items.append('<item id="front" href="images/front.jpg" media-type="image/jpeg"/>')
        items.append('<item id="cp" href="text/cover.xhtml" media-type="application/xhtml+xml"/>')
        spine.insert(0, '<itemref idref="cp"/>')
        guide = '<guide><reference type="cover" href="text/cover.xhtml"/></guide>'
    files["OEBPS/content.opf"] = (
        '<?xml version="1.0" encoding="utf-8"?>'
        '<package xmlns="http://www.idpf.org/2007/opf" version="3.0" unique-identifier="id">'
        '<metadata xmlns:dc="http://purl.org/dc/elements/1.1/">'
        '<dc:identifier id="id">urn:uuid:12345678-1234-1234-1234-123456789abc</dc:identifier>'
        f"<dc:title>{title}</dc:title><dc:creator>{author}</dc:creator>"
        "<dc:language>en</dc:language>"
        '<meta property="dcterms:modified">2026-01-01T00:00:00Z</meta></metadata>'
        f"<manifest>{''.join(items)}</manifest>"
        f"<spine{spine_attrs}>{''.join(spine)}</spine>{guide}</package>"
    )
    files["META-INF/container.xml"] = (
        '<?xml version="1.0"?><container version="1.0" '
        'xmlns="urn:oasis:names:tc:opendocument:xmlns:container"><rootfiles>'
        '<rootfile full-path="OEBPS/content.opf" media-type="application/oebps-package+xml"/>'
        "</rootfiles></container>"
    )
    files.update(extra_files or {})
    with zipfile.ZipFile(path, "w") as archive:
        archive.writestr(zipfile.ZipInfo("mimetype"), "application/epub+zip")
        for name, data in files.items():
            archive.writestr(name, data, compress_type=zipfile.ZIP_DEFLATED)
    return path

```python
"""EPUB 2/3 reader using only the standard library."""

import posixpath
import re
import zipfile
from html.parser import HTMLParser
from pathlib import Path
from urllib.parse import unquote, urldefrag
from xml.etree import ElementTree as ET

from lectr.book import Book, Chapter
from lectr.errors import DRM_MESSAGE, BookError
from lectr.readers.text import html_to_text

NS = {
    "c": "urn:oasis:names:tc:opendocument:xmlns:container",
    "opf": "http://www.idpf.org/2007/opf",
    "dc": "http://purl.org/dc/elements/1.1/",
    "ncx": "http://www.daisy.org/z3986/2005/ncx/",
}
_XMLENC_CIPHER_REF = "{http://www.w3.org/2001/04/xmlenc#}CipherReference"
_FONT_EXTENSIONS = (".ttf", ".otf", ".woff", ".woff2")
_HTML_TYPES = frozenset({"application/xhtml+xml", "text/html"})
_COVER_TYPES = frozenset({"image/jpeg", "image/png"})
_XML_ENCODING = re.compile(rb"""<\?xml[^>]*?encoding=["']([A-Za-z0-9._-]+)["']""")
_IMAGE_SOURCE = re.compile(r"""<(?:img|image)\b[^>]*?\b(?:src|xlink:href|href)=["']([^"']+)["']""")


def read_epub(path: Path) -> Book:
    try:
        with zipfile.ZipFile(path) as archive:
            return _read(archive)
    except (zipfile.BadZipFile, KeyError, ET.ParseError, OSError) as exc:
        raise BookError(f"{path.name}: not a valid EPUB ({exc}).") from exc


def parse_opf_metadata(opf: ET.Element) -> tuple[str, str]:
    def first(tag: str) -> str:
        element = opf.find(f"opf:metadata/dc:{tag}", NS)
        return " ".join((element.text or "").split()) if element is not None else ""

    return first("title"), first("creator")


def _read(archive: zipfile.ZipFile) -> Book:
    _reject_drm(archive)
    opf_path = _rootfile(archive)
    opf = ET.fromstring(archive.read(opf_path))
    base = posixpath.dirname(opf_path)
    manifest = {item.get("id", ""): item for item in opf.iterfind("opf:manifest/opf:item", NS)}
    title, author = parse_opf_metadata(opf)
    toc = _toc(archive, opf, manifest, base)

    chapters: list[tuple[str, list[str], bool]] = []  # (title, texts, came_from_toc)
    for ref in opf.iterfind("opf:spine/opf:itemref", NS):
        item = manifest.get(ref.get("idref", ""))
        if ref.get("linear") == "no" or item is None or item.get("media-type") not in _HTML_TYPES:
            continue
        if "nav" in (item.get("properties") or "").split():
            continue  # the table of contents page is navigation, not narration [A10]
        href = _resolve(base, item.get("href", ""))
        html = decode_markup(archive.read(href))
        for toc_title, section in _sections(html, toc.get(href, [])):
            parsed = html_to_text(section)
            if not parsed.text:
                continue  # e.g. an image-only cover page; must not use up a number [A10]
            # Text the TOC does not name, after a TOC-named chapter, continues that chapter
            # (e.g. Calibre splits long chapters into several files).
            if toc_title is None and toc and chapters and chapters[-1][2]:
                chapters[-1][1].append(parsed.text)
                continue
            name = toc_title or parsed.first_heading or f"Chapter {len(chapters) + 1}"
            chapters.append((name, [parsed.text], toc_title is not None))

    return Book(
        title,
        author,
        _cover(archive, opf, manifest, base),
        tuple(Chapter(t, "\n\n".join(texts)) for t, texts, _ in chapters),
    )


def decode_markup(data: bytes) -> str:
    """Decode XHTML/HTML using the encoding its XML prolog declares, else UTF-8. [A11]"""
    match = _XML_ENCODING.match(data.removeprefix(b"\xef\xbb\xbf"))
    encoding = match.group(1).decode("ascii") if match else "utf-8"
    try:
        return data.decode(encoding, errors="replace")
    except LookupError:
        return data.decode("utf-8", errors="replace")


def _sections(html: str, entries: list[tuple[str, str]]) -> list[tuple[str | None, str]]:
    """Cut one file at the TOC's fragment anchors; several chapters may share a file. [A4]"""
    cuts: dict[int, str] = {}
    for fragment, entry_title in entries:
        if not fragment:
            cuts.setdefault(0, entry_title)
            continue
        anchor = _anchor(html, fragment)
        if anchor >= 0:
            cuts.setdefault(html.rfind("<", 0, anchor), entry_title)
    if not cuts:
        return [(None, html)]
    starts = sorted(cuts)
    sections: list[tuple[str | None, str]] = []
    if starts[0] > 0:
        sections.append((None, html[: starts[0]]))
    for index, start in enumerate(starts):
        end = starts[index + 1] if index + 1 < len(starts) else len(html)
        sections.append((cuts[start], html[start:end]))
    return sections


def _anchor(html: str, fragment: str) -> int:
    for quote in ('"', "'"):
        position = html.find(f"id={quote}{fragment}{quote}")
        if position >= 0:
            return position
    return -1


def _reject_drm(archive: zipfile.ZipFile) -> None:
    names = set(archive.namelist())
    if "META-INF/rights.xml" in names:
        raise BookError(DRM_MESSAGE)
    if "META-INF/encryption.xml" in names:
        root = ET.fromstring(archive.read("META-INF/encryption.xml"))
        uris = [ref.get("URI", "") for ref in root.iter(_XMLENC_CIPHER_REF)]
        # Font obfuscation also uses encryption.xml; only encrypted content is DRM.
        if any(not unquote(uri).lower().endswith(_FONT_EXTENSIONS) for uri in uris):
            raise BookError(DRM_MESSAGE)


def _rootfile(archive: zipfile.ZipFile) -> str:
    container = ET.fromstring(archive.read("META-INF/container.xml"))
    rootfile = container.find("c:rootfiles/c:rootfile", NS)
    full_path = rootfile.get("full-path", "") if rootfile is not None else ""
    if not full_path:
        raise KeyError("container.xml has no rootfile")
    return full_path


def _resolve(base: str, href: str) -> str:
    return posixpath.normpath(posixpath.join(base, unquote(urldefrag(href).url)))


def _cover(
    archive: zipfile.ZipFile, opf: ET.Element, manifest: dict[str, ET.Element], base: str
) -> bytes | None:
    item = next(
        (i for i in manifest.values() if "cover-image" in (i.get("properties") or "").split()),
        None,
    )
    if item is None:
        meta = opf.find("opf:metadata/opf:meta[@name='cover']", NS)
        if meta is not None:
            item = manifest.get(meta.get("content", ""))
    if item is not None:
        if item.get("media-type") not in _COVER_TYPES:
            return None
        return _read_member(archive, _resolve(base, item.get("href", "")))
    return _guide_cover(archive, opf, base)


def _guide_cover(archive: zipfile.ZipFile, opf: ET.Element, base: str) -> bytes | None:
    """EPUB 2 books may name only a cover *page* in <guide>; use its first image. [A15]"""
    reference = opf.find("opf:guide/opf:reference[@type='cover']", NS)
    if reference is None:
        return None
    page = _resolve(base, reference.get("href", ""))
    html = _read_member(archive, page)
    image = _IMAGE_SOURCE.search(decode_markup(html)) if html is not None else None
    if image is None:
        return None
    return _read_member(archive, _resolve(posixpath.dirname(page), image.group(1)))


def _read_member(archive: zipfile.ZipFile, name: str) -> bytes | None:
    try:
        return archive.read(name)
    except KeyError:
        return None


def _toc(
    archive: zipfile.ZipFile, opf: ET.Element, manifest: dict[str, ET.Element], base: str
) -> dict[str, list[tuple[str, str]]]:
    """Map spine file path -> [(fragment, title)] from the EPUB 3 nav, else the NCX."""
    entries = _nav_entries(archive, manifest, base) or _ncx_entries(archive, opf, manifest, base)
    toc: dict[str, list[tuple[str, str]]] = {}
    for href, fragment, title in entries:
        if title and (fragment, title) not in toc.setdefault(href, []):
            toc[href].append((fragment, title))
    return toc


def _split_href(directory: str, href: str) -> tuple[str, str]:
    path, fragment = urldefrag(href)
    return _resolve(directory, path), unquote(fragment)


def _nav_entries(
    archive: zipfile.ZipFile, manifest: dict[str, ET.Element], base: str
) -> list[tuple[str, str, str]]:
    nav = next((i for i in manifest.values() if "nav" in (i.get("properties") or "").split()), None)
    if nav is None:
        return []
    nav_path = _resolve(base, nav.get("href", ""))
    parser = _NavParser()
    parser.feed(decode_markup(archive.read(nav_path)))
    parser.close()
    nav_dir = posixpath.dirname(nav_path)
    return [(*_split_href(nav_dir, href), title) for href, title in parser.entries]


def _ncx_entries(
    archive: zipfile.ZipFile, opf: ET.Element, manifest: dict[str, ET.Element], base: str
) -> list[tuple[str, str, str]]:
    spine = opf.find("opf:spine", NS)
    item = manifest.get(spine.get("toc", "") if spine is not None else "")
    if item is None:
        return []
    ncx_path = _resolve(base, item.get("href", ""))
    ncx = ET.fromstring(archive.read(ncx_path))
    ncx_dir = posixpath.dirname(ncx_path)
    entries = []
    for point in ncx.iter(f"{{{NS['ncx']}}}navPoint"):
        content = point.find("ncx:content", NS)
        label = point.find("ncx:navLabel/ncx:text", NS)
        if content is not None and label is not None:
            title = " ".join((label.text or "").split())
            entries.append((*_split_href(ncx_dir, content.get("src", "")), title))
    return entries


class _NavParser(HTMLParser):
    """Collect (href, text) for links inside <nav epub:type="toc">."""

    def __init__(self) -> None:
        super().__init__(convert_charrefs=True)
        self.entries: list[tuple[str, str]] = []
        self._navs: list[bool] = []
        self._href: str | None = None
        self._text: list[str] = []

    def handle_starttag(self, tag: str, attrs: list[tuple[str, str | None]]) -> None:
        attributes = dict(attrs)
        if tag == "nav":
            self._navs.append("toc" in (attributes.get("epub:type") or "").split())
        elif tag == "a" and any(self._navs):
            href = attributes.get("href")
            if href:
                self._href, self._text = href, []

    def handle_endtag(self, tag: str) -> None:
        if tag == "nav" and self._navs:
            self._navs.pop()
        elif tag == "a" and self._href is not None:
            self.entries.append((self._href, " ".join("".join(self._text).split())))
            self._href = None

    def handle_data(self, data: str) -> None:
        if self._href is not None:
            self._text.append(data)

from pypdf import PdfWriter
from pypdf.generic import DictionaryObject, NameObject, StreamObject


def make_pdf(
    path: Path,
    pages: Sequence[str],
    outline: Sequence[tuple[str, int] | tuple[str, int, str]] = (),
    *,
    title: str | None = None,
    author: str | None = None,
    user_password: str | None = None,
) -> Path:
    """Build a PDF with a real text layer (Helvetica), optional outline and encryption.

    outline: (title, page) entries, or (title, page, parent_title) for nested ones.
    user_password="" makes an owner-password-only (permissions) PDF.
    """
    writer = PdfWriter()
    font = writer._add_object(
        DictionaryObject(
            {
                NameObject("/Type"): NameObject("/Font"),
                NameObject("/Subtype"): NameObject("/Type1"),
                NameObject("/BaseFont"): NameObject("/Helvetica"),
            }
        )
    )
    for text in pages:
        page = writer.add_blank_page(612, 792)
        ops = ["BT", "/F1 12 Tf", "14 TL", "72 720 Td"]
        for line in text.split("\n"):
            escaped = line.replace("\\", "\\\\").replace("(", "\\(").replace(")", "\\)")
            ops.append(f"({escaped}) Tj T*")
        ops.append("ET")
        stream = StreamObject()
        stream.set_data("\n".join(ops).encode("latin-1"))
        page[NameObject("/Contents")] = writer._add_object(stream)
        page[NameObject("/Resources")] = DictionaryObject(
            {NameObject("/Font"): DictionaryObject({NameObject("/F1"): font})}
        )
    added: dict[str, object] = {}
    for entry in outline:
        parent = added[entry[2]] if len(entry) == 3 else None
        added[entry[0]] = writer.add_outline_item(entry[0], entry[1], parent=parent)
    metadata = {key: value for key, value in (("/Title", title), ("/Author", author)) if value}
    if metadata:
        writer.add_metadata(metadata)
    if user_password is not None:
        writer.encrypt(user_password=user_password, owner_password="owner", algorithm="AES-128")
    with path.open("wb") as handle:
        writer.write(handle)
    return path
```

```python
"""PDF reader: text layer via pypdf, chapters from top-level bookmarks."""

import logging
import re
from pathlib import Path

from pypdf import PageObject, PdfReader
from pypdf.errors import DependencyError, PdfReadError
from pypdf.generic import Destination

from lectr.book import Book, Chapter
from lectr.errors import DRM_MESSAGE, BookError

log = logging.getLogger(__name__)
_PAGE_NUMBER_LINE = re.compile(r"^[ \t]*\d{1,4}[ \t]*$", re.MULTILINE)


def read_pdf(path: Path) -> Book:
    try:
        reader = PdfReader(path)
        # Owner-password-only PDFs restrict printing/editing, not reading.
        if reader.is_encrypted and not reader.decrypt(""):
            raise BookError(DRM_MESSAGE)
        pages = [_page_text(page) for page in reader.pages]
        starts = _outline_starts(reader)
        metadata = reader.metadata
    except (PdfReadError, DependencyError, OSError, ValueError, KeyError) as exc:
        raise BookError(f"{path.name}: could not read PDF ({exc}).") from exc
    if not any(pages):
        raise BookError(f"{path.name}: no extractable text (scanned PDF?). OCR is not supported.")
    title = (metadata.title if metadata and metadata.title else "") or path.stem
    author = (metadata.author if metadata and metadata.author else "") or ""
    if not starts:
        log.warning(
            "%s has no bookmarks; it will be read as equal parts of about 30 minutes.", path.name
        )  # [A18]
    return Book(title, author, None, tuple(_chapters(pages, starts, title)))


def _page_text(page: PageObject) -> str:
    return _PAGE_NUMBER_LINE.sub("", page.extract_text() or "").strip()


def _outline_starts(reader: PdfReader) -> list[tuple[str, int]]:
    top = [item for item in reader.outline if isinstance(item, Destination)]
    nested = [
        item
        for group in reader.outline
        if isinstance(group, list)
        for item in group
        if isinstance(item, Destination)
    ]
    # A lone "Book" entry or a couple of "Part" entries usually wrap the real chapters,
    # so go one level down; parents still name any pages before their first child. [A5]
    entries = nested + top if nested and len(top) <= 2 else top
    starts: dict[int, str] = {}
    for item in entries:
        page = reader.get_destination_page_number(item)
        if page is not None and page >= 0:
            starts.setdefault(page, str(item.title or ""))
    return [(title, page) for page, title in sorted(starts.items())]


def _chapters(pages: list[str], starts: list[tuple[str, int]], title: str) -> list[Chapter]:
    # Pages join with "\n" so sentences and hyphenation continue across page breaks.
    if not starts:
        return [Chapter(title, "\n".join(pages))]
    chapters = []
    if starts[0][1] > 0:
        chapters.append(Chapter("Front Matter", "\n".join(pages[: starts[0][1]])))
    for index, (name, first_page) in enumerate(starts):
        end = starts[index + 1][1] if index + 1 < len(starts) else len(pages)
        chapters.append(Chapter(name, "\n".join(pages[first_page:end])))
    return chapters

```python
"""MOBI/AZW3 reader: unpack with the `mobi` library, then reuse the EPUB/PDF readers."""

import re
import shutil
from pathlib import Path
from xml.etree import ElementTree as ET

import mobi

from lectr.book import Book, Chapter
from lectr.errors import DRM_MESSAGE, BookError
from lectr.readers.epub import NS, parse_opf_metadata, read_epub
from lectr.readers.pdf import read_pdf
from lectr.readers.text import html_to_text

_PAGEBREAK = re.compile(r"<mbp:pagebreak\s*/?>", re.IGNORECASE)


def read_kindle(path: Path) -> Book:
    try:
        # ponytail: if unpacking fails midway, mobi.extract leaks its own temp folder (it
        # never returns the path); the OS temp cleaner reclaims it. Wrap tempfile.tempdir if
        # this ever matters. [G5]
        tempdir, extracted = mobi.extract(str(path))
    except Exception as exc:  # KindleUnpack raises assorted exception types
        if "encrypted" in str(exc).lower():
            raise BookError(DRM_MESSAGE) from exc
        raise BookError(f"{path.name}: could not unpack MOBI/AZW3 ({exc}).") from exc
    out = Path(extracted)
    try:
        if out.suffix.lower() == ".epub":
            book = read_epub(out)  # KF8 / AZW3
        elif out.suffix.lower() == ".pdf":
            book = read_pdf(out)  # Print Replica
        else:
            book = _read_mobi7(out)  # old MOBI: one book.html
    finally:
        shutil.rmtree(tempdir, ignore_errors=True)
    title = book.title if book.title and book.title != out.stem else path.stem
    return Book(title, book.author, book.cover, book.chapters)


def _read_mobi7(html_path: Path) -> Book:
    folder = html_path.parent
    html = html_path.read_text(encoding="utf-8", errors="replace")
    title, author = "", ""
    opf_path = folder / "content.opf"
    if opf_path.exists():
        title, author = parse_opf_metadata(ET.fromstring(opf_path.read_bytes()))
    cuts = _ncx_cuts(folder / "toc.ncx", html)
    if cuts:
        chapters = _split_at(html, cuts)
    else:
        chapters = []
        for number, segment in enumerate(_PAGEBREAK.split(html), 1):
            parsed = html_to_text(segment)
            chapters.append(Chapter(parsed.first_heading or f"Chapter {number}", parsed.text))
    return Book(title, author, None, tuple(chapters))


def _ncx_cuts(ncx_path: Path, html: str) -> list[tuple[int, str]]:
    """Positions in book.html where NCX entries start, found via their filepos anchors."""
    if not ncx_path.exists():
        return []
    ncx = ET.fromstring(ncx_path.read_bytes())
    cuts: dict[int, str] = {}
    for point in ncx.iter(f"{{{NS['ncx']}}}navPoint"):
        content = point.find("ncx:content", NS)
        label = point.find("ncx:navLabel/ncx:text", NS)
        if content is None or label is None:
            continue
        fragment = content.get("src", "").partition("#")[2]
        anchor = html.find(f'id="{fragment}"') if fragment else -1
        if anchor >= 0:
            cuts.setdefault(html.rfind("<", 0, anchor), " ".join((label.text or "").split()))
    return sorted(cuts.items())


def _split_at(html: str, cuts: list[tuple[int, str]]) -> list[Chapter]:
    chapters = [Chapter("Front Matter", html_to_text(html[: cuts[0][0]]).text)]
    for index, (start, title) in enumerate(cuts):
        end = cuts[index + 1][0] if index + 1 < len(cuts) else len(html)
        chapters.append(Chapter(title, html_to_text(html[start:end]).text))
    return chapters
```

"""Kokoro model files and speech synthesis."""

import hashlib
import os
import urllib.request
from collections.abc import Callable
from pathlib import Path
from typing import Any

import numpy as np
import platformdirs
from numpy.typing import NDArray

from lectr.errors import ModelError

SAMPLE_RATE = 24_000
MODEL_RELEASE_URL = (
    "https://github.com/thewh1teagle/kokoro-onnx/releases/download/model-files-v1.1"
)
MODEL_FILE = "kokoro-v1.0.onnx"
VOICES_FILE = "voices-v1.0.bin"
# Digests from the GitHub release metadata for model-files-v1.1.
MODEL_FILES: dict[str, str] = {
    MODEL_FILE: "beb0d1848dee9a49da392cc3df26958d46cfa35d321edf434f52949153f0df3a",
    VOICES_FILE: "bca610b8308e8d99f32e6fe4197e7ec01679264efed0cac9140fe9c29f1fbf7d",
}
# American (a*) and British (b*) English voices in voices-v1.0.bin.
ENGLISH_VOICES: tuple[str, ...] = (
    "af_alloy", "af_aoede", "af_bella", "af_heart", "af_jessica", "af_kore", "af_nicole",
    "af_nova", "af_river", "af_sarah", "af_sky", "am_adam", "am_echo", "am_eric", "am_fenrir",
    "am_liam", "am_michael", "am_onyx", "am_puck", "am_santa", "bf_alice", "bf_emma",
    "bf_isabella", "bf_lily", "bm_daniel", "bm_fable", "bm_george", "bm_lewis",
)

Audio = NDArray[np.float32]
SynthesizeFn = Callable[[str], Audio]
Progress = Callable[[str, int, int], None]


def default_model_dir() -> Path:
    return Path(platformdirs.user_data_dir("lectr", appauthor=False)) / "models"


def _open(url: str) -> Any:
    return urllib.request.urlopen(url, timeout=60)


def ensure_model(
    model_dir: Path, *, opener: Callable[[str], Any] = _open, progress: Progress | None = None
) -> tuple[Path, Path]:
    """Download any missing model file, verifying its SHA-256 before it is used."""
    model_dir.mkdir(parents=True, exist_ok=True)
    for name, digest in MODEL_FILES.items():
        target = model_dir / name
        if not target.exists():
            _download(f"{MODEL_RELEASE_URL}/{name}", target, digest, opener, progress)
    return model_dir / MODEL_FILE, model_dir / VOICES_FILE


def _download(
    url: str,
    target: Path,
    digest: str,
    opener: Callable[[str], Any],
    progress: Progress | None,
) -> None:
    partial = target.with_name(target.name + ".part")
    sha = hashlib.sha256()
    try:
        with opener(url) as response, partial.open("wb") as out:
            total = int(response.headers.get("Content-Length") or 0)
            done = 0
            while chunk := response.read(1 << 20):
                sha.update(chunk)
                out.write(chunk)
                done += len(chunk)
                if progress is not None:
                    progress(target.name, done, total)
    except OSError as exc:
        partial.unlink(missing_ok=True)
        raise ModelError(
            f"Could not download {target.name}: {exc}. Connect to the internet and run "
            f"`lectr model download`, or put the model files in {target.parent} "
            "(see --model-dir)."
        ) from exc
    if sha.hexdigest() != digest:
        partial.unlink(missing_ok=True)
        raise ModelError(f"{target.name} failed its checksum and was deleted; try again.")
    os.replace(partial, target)


class KokoroSynthesizer:
    """Speak text with one voice and speed; returns float32 samples at SAMPLE_RATE."""

    def __init__(self, model_path: Path, voices_path: Path, voice: str, speed: float) -> None:
        # Imported here: loading ONNX Runtime is slow and only `convert` needs it.
        from kokoro_onnx import Kokoro

        self._kokoro = Kokoro(str(model_path), str(voices_path))
        if voice not in self._kokoro.get_voices():
            raise ModelError(f"Voice {voice!r} is not in the model's voice file.")
        self._voice = voice
        self._speed = speed
        self._lang = "en-gb" if voice.startswith("b") else "en-us"

    def __call__(self, text: str) -> Audio:
        samples, rate = self._kokoro.create(
            text, voice=self._voice, speed=self._speed, lang=self._lang
        )
        if rate != SAMPLE_RATE:
            raise ModelError(f"Model returned {rate} Hz audio; expected {SAMPLE_RATE} Hz.")
        return samples
```

    voice: str = "af_heart"
    speed: float = 1.0
    format: str = "m4b"
    bitrate: int = 64
    max_part_hours: float = 12.0
    output_dir: Path | None = None
    cache_dir: Path | None = None
    model_dir: Path | None = None
    keep_cache: bool = False
    overwrite: bool = False

    directory = platformdirs.user_config_dir("lectr", appauthor=False, roaming=True)
    return Path(directory) / "config.yaml"


def _invalid(key: str, value: object, expected: str) -> ConfigError:
    return ConfigError(f"Invalid {key}: {value!r} (expected {expected}).")


def _number(
    kind: type[int] | type[float], low: float, high: float | None, expected: str
) -> Validator:
    """Accept numbers or numeric strings in [low, high]; with high=None the bound is (low, ∞)."""

    def check(key: str, value: object) -> object:
        if isinstance(value, bool) or not isinstance(value, int | float | str):
            raise _invalid(key, value, expected)
        if kind is int and isinstance(value, float) and not value.is_integer():
            raise _invalid(key, value, expected)
        try:
            number = kind(value)
        except ValueError:
            raise _invalid(key, value, expected) from None
        if not math.isfinite(number):  # nan/inf slip past range checks [A14]
            raise _invalid(key, value, expected)
        too_low = number <= low if high is None else number < low
        if too_low or (high is not None and number > high):
            raise _invalid(key, value, expected)
        return number

    return check


def _choice(options: tuple[str, ...]) -> Validator:
    def check(key: str, value: object) -> object:
        text = str(value).strip().lower()
        if text not in options:
            raise _invalid(key, value, " or ".join(options))
        return text

    return check


def _voice(key: str, value: object) -> object:
    text = str(value).strip()
    if text not in ENGLISH_VOICES:
        raise _invalid(key, value, "an English voice; run `lectr voices`")
    return text


def _bool(key: str, value: object) -> object:
    if isinstance(value, bool):
        return value
    text = str(value).strip().lower()
    if text in {"y", "yes", "true"}:
        return True
    if text in {"n", "no", "false"}:
        return False
    raise _invalid(key, value, "y or n")


def _path(key: str, value: object) -> object:
    if value is None or (isinstance(value, str) and value.strip().lower() in {"", "none", "null"}):
        return None
    if isinstance(value, str | Path):
        return Path(value).expanduser()
    raise _invalid(key, value, "a folder path or none")


_VALIDATORS: dict[str, Validator] = {
    "voice": _voice,
    "speed": _number(float, 0.5, 2.0, "a number from 0.5 to 2.0"),
    "format": _choice(FORMATS),
    "bitrate": _number(int, 16, 320, "a whole number from 16 to 320"),
    "max_part_hours": _number(float, 0, None, "a number of hours above 0"),
    "output_dir": _path,
    "cache_dir": _path,
    "model_dir": _path,
    "keep_cache": _bool,
    "overwrite": _bool,
}


def validate(key: str, value: object) -> object:
    validator = _VALIDATORS.get(key)
    if validator is None:
        raise ConfigError(f"Unknown setting '{key}'. Valid settings: {', '.join(KEYS)}.")
    return validator(key, value)


def load_file(path: Path) -> dict[str, object]:
    if not path.exists():
        return {}
    try:
        data = yaml.safe_load(path.read_text(encoding="utf-8"))
        if data is None:
            return {}
        if not isinstance(data, dict):
            raise ConfigError("expected 'key: value' settings")
        return {str(key): validate(str(key), value) for key, value in data.items()}
    except yaml.YAMLError as exc:
        raise ConfigError(f"{path}: invalid YAML ({exc})") from exc
    except ConfigError as exc:
        raise ConfigError(f"{path}: {exc}") from exc


def resolve(
    file_values: Mapping[str, object], flag_values: Mapping[str, object | None]
) -> tuple[Settings, dict[str, str]]:
    values: dict[str, Any] = {}  # Any: dataclasses.replace needs per-field types [G3]
    sources: dict[str, str] = {}
    for key in KEYS:
        flag = flag_values.get(key)
        if flag is not None:
            values[key], sources[key] = validate(key, flag), "flag"
        elif key in file_values:
            values[key], sources[key] = file_values[key], "file"
        else:
            sources[key] = "default"
    return dataclasses.replace(Settings(), **values), sources


def display(value: object) -> str:
    if value is None:
        return "none"
    if isinstance(value, bool):
        return "y" if value else "n"
    return str(value)


def save(path: Path, settings: Settings) -> None:
    data = {}
    for key in KEYS:
        value = getattr(settings, key)
        data[key] = str(value) if isinstance(value, Path) else value
    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_name(path.name + ".tmp")
    temporary.write_text(_HEADER + yaml.safe_dump(data, sort_keys=False), encoding="utf-8")
    os.replace(temporary, path)

```python
"""`lectr setup`: ask for each setting and write config.yaml."""

import dataclasses
from collections.abc import Callable
from pathlib import Path
from typing import Any

from lectr.config import KEYS, Settings, display, save, validate
from lectr.errors import ConfigError

PROMPTS: dict[str, str] = {
    "voice": "Voice (run `lectr voices` to list)",
    "speed": "Speaking speed, 0.5-2.0",
    "format": "Output format (m4b/mp3)",
    "bitrate": "Bitrate in kbps (16-320)",
    "max_part_hours": "Split M4B books longer than this many hours",
    "output_dir": "Output folder ('none' = next to the book)",
    "cache_dir": "Cache folder ('none' = system default)",
    "model_dir": "Model folder ('none' = system default)",
    "keep_cache": "Keep chapter audio after a successful run (y/n)",
    "overwrite": "Replace existing audiobooks (y/n)",
}


def run_wizard(
    path: Path,
    current: Settings,
    *,
    ask: Callable[[str], str],
    say: Callable[[str], None],
    offer_download: Callable[[], object] | None = None,
) -> bool:
    say(f"lectr setup: press Enter to keep the value in [brackets]. Saving to {path}")
    try:
        values: dict[str, Any] = {  # Any: dataclasses.replace needs per-field types [G3]
            key: _ask_value(key, getattr(current, key), ask, say) for key in KEYS
        }
        chosen = dataclasses.replace(current, **values)
        say("")
        say("Summary:")
        for key in KEYS:
            say(f"  {key}: {display(getattr(chosen, key))}")
        if not _yes(ask("Save these settings? (y/n) [y]: ")):
            say("Not saved.")
            return False
    except (KeyboardInterrupt, EOFError):
        say("")
        say("Setup cancelled; nothing was written.")
        return False
    save(path, chosen)
    say(f"Saved {path}")
    if offer_download is not None and _yes(
        ask("Download the voice model now (~350 MB)? (y/n) [y]: ")
    ):
        offer_download()
    return True


def _ask_value(
    key: str, current: object, ask: Callable[[str], str], say: Callable[[str], None]
) -> object:
    while True:
        answer = ask(f"{PROMPTS[key]} [{display(current)}]: ").strip()
        if not answer:
            return current
        try:
            return validate(key, answer)
        except ConfigError as exc:
            say(f"  {exc}")


def _yes(answer: str) -> bool:
    return answer.strip().lower() in {"", "y", "yes"}

"""Chapter WAV files, M4B part planning, and ffmpeg encoding."""

import os
import re
import shutil
import wave
from collections.abc import Sequence
from dataclasses import dataclass
from pathlib import Path

import numpy as np

from lectr.errors import AudioError
from lectr.tts import SAMPLE_RATE, Audio

FFMPEG_HINT = (
    "ffmpeg is required. Install it with `brew install ffmpeg` (macOS/Linux Homebrew), "
    "`sudo apt install ffmpeg` (Debian/Ubuntu), or `winget install ffmpeg` (Windows)."
)
_ILLEGAL_FILENAME = re.compile(r'[<>:"/\\|?*\x00-\x1f]')
_RESERVED_NAMES = frozenset(
    {
        "CON",
        "PRN",
        "AUX",
        "NUL",
        *(f"COM{i}" for i in range(1, 10)),
        *(f"LPT{i}" for i in range(1, 10)),
    }
)


@dataclass(frozen=True)
class ChapterAudio:
    number: int  # 1-based position in the book
    title: str
    path: Path
    seconds: float


@dataclass(frozen=True)
class Tags:
    title: str
    author: str
    album: str


def write_wav(path: Path, samples: Audio, *, trailing_silence: float = 1.0) -> None:
    """Write 16-bit mono PCM atomically: only a finished file gets the final name."""
    silence = np.zeros(int(SAMPLE_RATE * trailing_silence), dtype=np.float32)
    pcm = (np.concatenate([np.clip(samples, -1.0, 1.0), silence]) * 32767).astype("<i2")
    partial = path.with_name(path.name + ".part")
    with wave.open(str(partial), "wb") as out:
        out.setnchannels(1)
        out.setsampwidth(2)
        out.setframerate(SAMPLE_RATE)
        out.writeframes(pcm.tobytes())
    os.replace(partial, path)


def wav_seconds(path: Path) -> float:
    with wave.open(str(path), "rb") as wav:
        return wav.getnframes() / wav.getframerate()


def plan_parts(chapters: Sequence[ChapterAudio], max_hours: float) -> list[list[ChapterAudio]]:
    """Group chapters in order; start a new part when the next chapter would pass the limit."""
    limit = max_hours * 3600
    parts: list[list[ChapterAudio]] = []
    current: list[ChapterAudio] = []
    total = 0.0
    for item in chapters:
        if current and total + item.seconds > limit:
            parts.append(current)
            current, total = [], 0.0
        current.append(item)
        total += item.seconds
    if current:
        parts.append(current)
    return parts


def escape_metadata(value: str) -> str:
    value = " ".join(value.split())
    for special in "\\=;#":  # backslash first so later escapes are not doubled
        value = value.replace(special, "\\" + special)
    return value


def ffmetadata(tags: Tags, chapters: Sequence[ChapterAudio]) -> str:
    lines = [
        ";FFMETADATA1",
        f"title={escape_metadata(tags.title)}",
        f"artist={escape_metadata(tags.author)}",
        f"album={escape_metadata(tags.album)}",
        "genre=Audiobook",
    ]
    elapsed = 0.0
    for item in chapters:
        start = round(elapsed * 1000)
        elapsed += item.seconds
        lines += [
            "",
            "[CHAPTER]",
            "TIMEBASE=1/1000",
            f"START={start}",
            f"END={round(elapsed * 1000)}",
            f"title={escape_metadata(item.title)}",
        ]
    return "\n".join(lines) + "\n"


def sanitize_filename(name: str, max_length: int = 120) -> str:
    cleaned = _ILLEGAL_FILENAME.sub("_", " ".join(name.split()))[:max_length]
    cleaned = cleaned.strip().rstrip(" .")
    if not cleaned:
        return "untitled"
    if cleaned.split(".")[0].upper() in _RESERVED_NAMES:
        cleaned = "_" + cleaned
    return cleaned


def cover_suffix(data: bytes) -> str | None:
    if data.startswith(b"\xff\xd8\xff"):
        return ".jpg"
    if data.startswith(b"\x89PNG\r\n\x1a\n"):
        return ".png"
    return None


def m4b_paths(base: Path, count: int) -> list[Path]:
    if count == 1:
        return [base.with_name(base.name + ".m4b")]
    return [base.with_name(f"{base.name} - Part {i} of {count}.m4b") for i in range(1, count + 1)]


def find_ffmpeg() -> str:
    executable = shutil.which("ffmpeg")
    if executable is None:
        raise AudioError(FFMPEG_HINT)
    return executable
```

```python
_QUIET = ["-nostdin", "-hide_banner", "-loglevel", "error", "-y"]


def _run_ffmpeg(args: list[str], partial: Path) -> None:
    try:
        result = subprocess.run(
            args, capture_output=True, text=True, encoding="utf-8", errors="replace", check=False
        )
    except BaseException:  # Ctrl-C mid-encode: never leave a half-written file behind [G4]
        partial.unlink(missing_ok=True)
        raise
    if result.returncode != 0:
        partial.unlink(missing_ok=True)
        tail = "\n".join(result.stderr.strip().splitlines()[-20:])
        raise AudioError(f"ffmpeg failed (exit {result.returncode}):\n{tail}")


def _arg(path: Path) -> str:
    """Absolute path for ffmpeg: a name starting with '-' or 'proto:' is never an option
    or protocol. [A6]"""
    return str(path.absolute())


def _concat_line(path: Path) -> str:
    # ffmpeg concat syntax: inside single quotes, a quote is written as '\''
    return "file '" + path.resolve().as_posix().replace("'", "'\\''") + "'\n"


def encode_m4b(
    ffmpeg: str,
    chapters: Sequence[ChapterAudio],
    output: Path,
    tags: Tags,
    cover: Path | None,
    bitrate: int,
    work_dir: Path,
) -> None:
    concat = work_dir / "concat.txt"
    concat.write_text("".join(_concat_line(c.path) for c in chapters), encoding="utf-8")
    metadata = work_dir / "metadata.txt"
    metadata.write_text(ffmetadata(tags, chapters), encoding="utf-8")
    partial = output.with_name(output.name + ".partial")
    args = [ffmpeg, *_QUIET, "-f", "concat", "-safe", "0", "-i", _arg(concat)]
    args += ["-i", _arg(metadata)]
    if cover is not None:
        args += ["-i", _arg(cover)]
    args += ["-map", "0:a"]
    if cover is not None:
        args += ["-map", "2:v"]
    args += ["-map_metadata", "1", "-map_chapters", "1"]
    args += ["-c:a", "aac", "-b:a", f"{bitrate}k", "-ac", "1"]
    if cover is not None:
        args += ["-c:v", "copy", "-disposition:v:0", "attached_pic"]
    args += ["-movflags", "+faststart", "-f", "mp4", _arg(partial)]
    _run_ffmpeg(args, partial)
    os.replace(partial, output)


def encode_mp3(
    ffmpeg: str,
    chapter: ChapterAudio,
    output: Path,
    tags: Tags,
    track: str,
    cover: Path | None,
    bitrate: int,
) -> None:
    partial = output.with_name(output.name + ".partial")
    args = [ffmpeg, *_QUIET, "-i", _arg(chapter.path)]
    if cover is not None:
        args += ["-i", _arg(cover), "-map", "0:a", "-map", "1:v", "-c:v", "copy"]
        args += ["-disposition:v:0", "attached_pic"]
        args += ["-metadata:s:v", "title=Cover", "-metadata:s:v", "comment=Cover (front)"]
    args += ["-c:a", "libmp3lame", "-b:a", f"{bitrate}k", "-id3v2_version", "3"]
    args += ["-metadata", f"title={tags.title}", "-metadata", f"artist={tags.author}"]
    args += ["-metadata", f"album={tags.album}", "-metadata", f"track={track}"]
    args += ["-f", "mp3", _arg(partial)]
    _run_ffmpeg(args, partial)
    os.replace(partial, output)

_RANGE = re.compile(r"(\d+)(?:-(\d+))?")


@dataclass(frozen=True)
class ConvertRequest:
    book_path: Path
    output: Path | None = None
    chapters: str | None = None
    title: str | None = None
    author: str | None = None
    cover: Path | None = None
    resume: bool = True


def parse_chapter_selection(spec: str | None, total: int) -> list[int]:
    if spec is None:
        return list(range(1, total + 1))
    chosen: set[int] = set()
    for part in spec.split(","):
        match = _RANGE.fullmatch(part.strip())
        if match is None:
            raise LectrError(f"Invalid chapter range {part.strip()!r}; use a form like 1-3,7.")
        low = int(match.group(1))
        high = int(match.group(2) or low)
        if low < 1 or high > total or low > high:
            raise LectrError(f"Chapter range {part.strip()!r} is outside 1-{total}.")
        chosen.update(range(low, high + 1))
    return sorted(chosen)


def cache_key(book_path: Path, voice: str, speed: float) -> str:
    digest = hashlib.sha256()
    with book_path.open("rb") as handle:
        for block in iter(lambda: handle.read(1 << 20), b""):
            digest.update(block)
    digest.update(f"\0{voice}\0{speed}\0{__version__}".encode())
    return digest.hexdigest()[:16]


def default_cache_dir() -> Path:
    return Path(platformdirs.user_cache_dir("lectr", appauthor=False))


def output_base(request: ConvertRequest, settings: Settings, title: str) -> Path:
    """M4B: output path without extension. MP3: the folder that receives chapter files."""
    name = sanitize_filename(title)
    if request.output is not None:
        suffix = request.output.suffix.lower()
        if suffix == ".m4b" and settings.format == "m4b":
            return request.output.with_suffix("")
        if suffix in {".m4b", ".mp3"}:
            raise LectrError("For --format mp3, --output must be a folder.")
        return request.output / name
    return (settings.output_dir or request.book_path.parent) / name


def existing_outputs(base: Path, fmt: str) -> list[Path]:
    if fmt == "mp3":
        return [base] if base.is_dir() and any(base.iterdir()) else []
    single = base.with_name(base.name + ".m4b")
    parts = sorted(base.parent.glob(glob.escape(base.name) + " - Part * of *.m4b"))
    return [path for path in [single, *parts] if path.exists()]


def format_duration(seconds: float) -> str:
    hours, rest = divmod(round(seconds), 3600)
    minutes, secs = divmod(rest, 60)
    if hours:
        return f"{hours}h{minutes:02d}m{secs:02d}s"
    return f"{minutes}m{secs:02d}s" if minutes else f"{secs}s"


def synthesize_chapters(
    chapters: Sequence[tuple[int, Chapter]],
    work_dir: Path,
    synthesize: SynthesizeFn,
    report: Report,
) -> list[ChapterAudio]:
    """Synthesize each chapter once; a finished NNN.wav is reused on later runs."""
    work_dir.mkdir(parents=True, exist_ok=True)
    paths = {number: work_dir / f"{number:03d}.wav" for number, _ in chapters}
    cached = sum(path.exists() for path in paths.values())
    if cached:
        report(f"Resuming: {cached}/{len(chapters)} chapters cached")
    results = []
    for position, (number, chapter) in enumerate(chapters, 1):
        path = paths[number]
        if not path.exists():
            # Announce before synthesizing: a long chapter can take many minutes. [A17]
            report(f"[{position}/{len(chapters)}] {chapter.title} …")
            started = time.monotonic()
            write_wav(path, synthesize(chapter.text))
            report(
                f"    {format_duration(wav_seconds(path))} audio in "
                f"{format_duration(time.monotonic() - started)}"
            )
        results.append(ChapterAudio(number, chapter.title, path, wav_seconds(path)))
    return results


def convert(
    request: ConvertRequest,
    settings: Settings,
    *,
    make_synthesizer: Callable[[], SynthesizeFn],
    report: Report,
) -> list[Path]:
    book = read_book(request.book_path)
    title = request.title or book.title
    author = request.author or book.author
    numbers = parse_chapter_selection(request.chapters, len(book.chapters))
    # Cheap checks first, so a mistake never costs hours of synthesis.
    ffmpeg = find_ffmpeg()
    base = output_base(request, settings, title)
    existing = existing_outputs(base, settings.format)
    if existing and not settings.overwrite:
        raise LectrError(f"Output already exists: {existing[0]}. Use --overwrite to replace it.")
    cover_bytes = _read_cover(request.cover) if request.cover is not None else book.cover
    target_dir = base if settings.format == "mp3" else base.parent
    try:
        target_dir.mkdir(parents=True, exist_ok=True)
    except OSError as exc:  # e.g. -o names an existing file [A7]
        raise LectrError(f"Cannot use output folder {target_dir}: {exc}") from exc

    key = cache_key(request.book_path, settings.voice, settings.speed)
    work_dir = (settings.cache_dir or default_cache_dir()) / key
    if not request.resume:
        shutil.rmtree(work_dir, ignore_errors=True)
    work_dir.mkdir(parents=True, exist_ok=True)
    selected = [(number, book.chapters[number - 1]) for number in numbers]
    synthesizer: SynthesizeFn | None = None

    def synthesize(text: str) -> Audio:
        # Load the model only on the first uncached chapter; reruns after an encode
        # failure need no model at all. [A16]
        nonlocal synthesizer
        if synthesizer is None:
            synthesizer = make_synthesizer()
        return synthesizer(text)

    audio = synthesize_chapters(selected, work_dir, synthesize, report)
    cover = _cover_file(cover_bytes, work_dir)
    if settings.format == "m4b":
        outputs = _assemble_m4b(
            ffmpeg, audio, base, title, author, cover, settings, work_dir, report
        )
        for stale in set(existing) - set(outputs):
            stale.unlink(missing_ok=True)
    else:
        outputs = _assemble_mp3(
            ffmpeg, audio, len(book.chapters), base, title, author, cover, settings, report
        )
        if existing:  # --overwrite: drop files from earlier runs, e.g. a --chapters sample [A9]
            for stale in set(base.glob("*.mp3")) - set(outputs):
                stale.unlink()
    if not settings.keep_cache:
        shutil.rmtree(work_dir, ignore_errors=True)
    return outputs


def _read_cover(path: Path) -> bytes:
    try:
        return path.read_bytes()
    except OSError as exc:
        raise LectrError(f"Cannot read cover image {path}: {exc}") from exc


def _cover_file(data: bytes | None, work_dir: Path) -> Path | None:
    if data is None:
        return None
    suffix = cover_suffix(data)
    if suffix is None:
        log.warning("The cover image is not JPEG or PNG; the audiobook will have no cover.")
        return None
    path = work_dir / f"cover{suffix}"
    path.write_bytes(data)
    return path


def _assemble_m4b(
    ffmpeg: str,
    audio: list[ChapterAudio],
    base: Path,
    title: str,
    author: str,
    cover: Path | None,
    settings: Settings,
    work_dir: Path,
    report: Report,
) -> list[Path]:
    limit = settings.max_part_hours * 3600
    for item in audio:
        if item.seconds > limit:
            log.warning(
                "Chapter %d (%s) is longer than %g hours; it becomes a part of its own.",
                item.number,
                item.title,
                settings.max_part_hours,
            )
    parts = plan_parts(audio, settings.max_part_hours)
    paths = m4b_paths(base, len(parts))
    for index, (part, path) in enumerate(zip(parts, paths, strict=True), 1):
        album = title if len(parts) == 1 else f"{title} - Part {index} of {len(parts)}"
        report(f"Encoding {path.name}")
        encode_m4b(
            ffmpeg, part, path, Tags(album, author, album), cover, settings.bitrate, work_dir
        )
    return paths


def _assemble_mp3(
    ffmpeg: str,
    audio: list[ChapterAudio],
    total_chapters: int,
    folder: Path,
    title: str,
    author: str,
    cover: Path | None,
    settings: Settings,
    report: Report,
) -> list[Path]:
    # Files and tracks are numbered by chapter, so a sample run and a full run agree. [A9]
    width = max(2, len(str(total_chapters)))
    outputs = []
    for item in audio:
        path = folder / f"{item.number:0{width}d} - {sanitize_filename(item.title, 80)}.mp3"
        report(f"Encoding {path.name}")
        encode_mp3(
            ffmpeg,
            item,
            path,
            Tags(item.title, author, title),
            f"{item.number}/{total_chapters}",
            cover,
            settings.bitrate,
        )
        outputs.append(path)
    return outputs
```

import argparse
import io
import logging
import sys
from collections.abc import Sequence
from pathlib import Path

from lectr import __version__
from lectr.config import FORMATS, KEYS, Settings, default_config_path, display, load_file, resolve
from lectr.errors import LectrError

def build_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(
        prog="lectr",
        description="Turn DRM-free e-books into audiobooks with a local text-to-speech model.",
    )
    parser.add_argument("--version", action="version", version=f"lectr {__version__}")
    common = argparse.ArgumentParser(add_help=False)
    common.add_argument("--config", type=Path, help="config file (default: OS config folder)")
    common.add_argument("-q", "--quiet", action="store_true", help="hide progress output")
    common.add_argument("-v", "--verbose", action="store_true", help="show debug logging")
    commands = parser.add_subparsers(dest="command", required=True)

    conv = commands.add_parser("convert", parents=[common], help="convert a book to an audiobook")
    conv.add_argument("book", type=Path)
    conv.add_argument("-o", "--output", type=Path, help="output folder, or a .m4b file path")
    conv.add_argument("--format", choices=FORMATS)
    conv.add_argument("--max-part-hours", dest="max_part_hours", type=float)
    conv.add_argument("--bitrate", type=int, help="kbps (16-320)")
    conv.add_argument("--voice", help="see `lectr voices`")
    conv.add_argument("--speed", type=float, help="0.5-2.0")
    conv.add_argument("--chapters", metavar="RANGE", help="e.g. 1-3,7")
    conv.add_argument("--title")
    conv.add_argument("--author")
    conv.add_argument("--cover", type=Path, help="JPEG or PNG image")
    conv.add_argument("--cache-dir", dest="cache_dir", type=Path)
    conv.add_argument("--model-dir", dest="model_dir", type=Path)
    conv.add_argument("--no-resume", action="store_true", help="ignore cached chapters")
    conv.add_argument("--keep-cache", dest="keep_cache", action="store_true", default=None)
    conv.add_argument("--overwrite", action="store_true", default=None)

    commands.add_parser("setup", parents=[common], help="choose and save default settings")
    config = commands.add_parser("config", help="inspect settings")
    config_commands = config.add_subparsers(dest="config_command", required=True)
    config_commands.add_parser(
        "show", parents=[common], help="print effective settings and their sources"
    )
    commands.add_parser("voices", help="list available voices")
    chapters = commands.add_parser("chapters", help="list the chapters lectr detects")
    chapters.add_argument("book", type=Path)
    model = commands.add_parser("model", help="manage the voice model")
    model_commands = model.add_subparsers(dest="model_command", required=True)
    download = model_commands.add_parser(
        "download", parents=[common], help="download and verify the model files"
    )
    download.add_argument("--model-dir", dest="model_dir", type=Path)
    return parser


def main(argv: Sequence[str] | None = None) -> int:
    # Windows pipes default to the ANSI code page; never crash printing a non-ASCII title [G6]
    for stream in (sys.stdout, sys.stderr):
        if isinstance(stream, io.TextIOWrapper):
            stream.reconfigure(errors="backslashreplace")
    args = build_parser().parse_args(argv)
    logging.basicConfig(
        level=logging.DEBUG if getattr(args, "verbose", False) else logging.WARNING,
        format="%(levelname)s: %(message)s",
    )
    try:
        return _dispatch(args)
    except KeyboardInterrupt:
        hint = ""
        if args.command == "convert":
            hint = " Progress saved; rerun the same command to resume."
        print(f"\nInterrupted.{hint}", file=sys.stderr)
        return 130
    except (LectrError, OSError) as exc:  # OSError: disk full, file locked by a player [A19]
        print(f"error: {exc}", file=sys.stderr)
        return 1


def _dispatch(args: argparse.Namespace) -> int:
    if args.command == "voices":
        print("\n".join(ENGLISH_VOICES))
        return 0
    if args.command == "chapters":
        for number, chapter in enumerate(read_book(args.book).chapters, 1):
            print(f"{number:3d}. {chapter.title} ({len(chapter.text.split())} words)")
        return 0
    config_path: Path = args.config or default_config_path()
    flags = {key: getattr(args, key, None) for key in KEYS}
    settings, sources = resolve(load_file(config_path), flags)
    report = _reporter(args.quiet)
    if args.command == "convert":
        return _convert(args, settings, config_path, report)
    if args.command == "setup":
        run_wizard(
            config_path,
            settings,
            ask=input,
            say=print,
            offer_download=lambda: _ensure_model(settings, args.quiet),
        )
        return 0
    if args.command == "config":
        status = "" if config_path.exists() else " (not created yet; run `lectr setup`)"
        print(f"Config file: {config_path}{status}")
        for key in KEYS:
            print(f"{key}: {display(getattr(settings, key))} ({sources[key]})")
        return 0
    _ensure_model(settings, args.quiet)  # `model download`
    return 0


def _convert(
    args: argparse.Namespace, settings: Settings, config_path: Path, report: Report
) -> int:
    if not config_path.exists():
        report("Tip: run `lectr setup` to save your preferred voice, format, and folders.")
    request = ConvertRequest(
        book_path=args.book,
        output=args.output,
        chapters=args.chapters,
        title=args.title,
        author=args.author,
        cover=args.cover,
        resume=not args.no_resume,
    )

    def make_synthesizer() -> SynthesizeFn:
        model_path, voices_path = _ensure_model(settings, args.quiet)
        return KokoroSynthesizer(model_path, voices_path, settings.voice, settings.speed)

    for path in convert(request, settings, make_synthesizer=make_synthesizer, report=report):
        print(path)
    return 0


def _ensure_model(settings: Settings, quiet: bool) -> tuple[Path, Path]:
    return ensure_model(
        settings.model_dir or default_model_dir(), progress=None if quiet else _show_download
    )


def _show_download(name: str, done: int, total: int) -> None:
    if total:
        print(f"\rDownloading {name}: {done * 100 // total}%", end="", file=sys.stderr, flush=True)
        if done >= total:
            print(file=sys.stderr)


def _reporter(quiet: bool) -> Report:
    def report(message: str) -> None:
        if not quiet:
            print(message, file=sys.stderr)

    return report

```md
# Architecture

lectr is a single-process CLI with a three-stage pipeline.

    book file ─► readers/ ─► Book(chapters) ─► pipeline.synthesize_chapters ─► cache/NNN.wav
                                                                │
                                              audio.encode_m4b / encode_mp3 ─► .m4b parts / .mp3 files

## Components

| Module | Responsibility | Depends on |
|---|---|---|
| `readers/` | One reader per format → `Book`; `read_book` dispatches by extension and cleans text | stdlib, pypdf, mobi |
| `tts.py` | Verified model download; `KokoroSynthesizer` (text → float32 samples at 24 kHz) | kokoro-onnx, platformdirs |
| `pipeline.py` | Chapter selection, cache key, resume, output naming, orchestration | readers, tts, audio, config |
| `audio.py` | WAV I/O, part planning, FFMETADATA, ffmpeg invocation | numpy, ffmpeg (external) |
| `config.py` / `wizard.py` | Settings: flag > YAML file > default; interactive setup | PyYAML, platformdirs |
| `cli.py` | argparse subcommands, exit codes, progress output | all of the above |

## Key properties

- **Resumable:** each chapter is written to `NNN.wav.part` and renamed when complete; reruns skip finished chapters. The cache key is sha256(book bytes, voice, speed, lectr version).
- **Fail fast:** format, chapter range, ffmpeg presence, and output collisions are checked before the model loads.
- **No shell:** ffmpeg runs with argument lists; metadata goes through an escaped FFMETADATA file.
- **Offline after setup:** only the one-time model download uses the network; files are verified against pinned SHA-256 digests.

## Trade-offs

- Whole chapters are synthesized in one call (kokoro-onnx batches internally); progress is per chapter.
- PDF text quality depends on the PDF; running headers and footers are not removed.
- One TTS engine; no plugin interface until a second engine is needed.

Decisions: see `docs/adr/`.
```

```md
# ADR-003: Per-chapter WAV cache with resume

- **Date**: 2026-09-26
- **Status**: Accepted

## Context

Synthesizing a long book on CPU takes hours; interruptions must not lose finished work, and M4B parts need known chapter durations.

## Decision

Synthesize each chapter to `cache/<key>/NNN.wav` through an atomic rename, then assemble with ffmpeg; delete the cache after success unless `--keep-cache`.

## Rationale

Reruns resume at chapter granularity; durations are known before encoding; each stage is testable without the model.

## Consequences

- **Positive**: resumable, testable, simple.
- **Negative**: temporary disk use (~170 MB per hour of audio).
- **Follow-ups**: parallel chapter synthesis only if measurements justify it.

## Alternatives Considered

- Single streaming pass — loses all work on interruption and cannot plan parts.
- SQLite job store with sentence checkpoints — more machinery for little gain.
```

# ADR-004: Split long M4B output at chapter boundaries (default 12 hours)

- **Date**: 2026-09-26
- **Status**: Accepted

## Context

Many players fail on very long M4B files: 32-bit sample counters overflow at ~49.7 hours for 24 kHz audio, and users report failures near 50 hours.

## Decision

Group chapters greedily into parts no longer than `max_part_hours` (default 12); a single longer chapter becomes its own part with a warning. Files are named `Title - Part i of N.m4b`.

## Rationale

Well under player limits while keeping most novels in one file; never cuts mid-chapter.

## Consequences

- **Positive**: broad player compatibility.
- **Negative**: very long books become several files.
- **Follow-ups**: none.

## Alternatives Considered

- 8-hour parts (old-iPod safe) — splits most novels.
- No splitting — risks unplayable files.
```

class Lectr < Formula
  desc "Turn DRM-free e-books into audiobooks with a local text-to-speech model"
  homepage "https://github.com/rgoshen/lectr"
  url "https://files.pythonhosted.org/packages/source/l/lectr/lectr-{{VERSION}}.tar.gz"
  sha256 "{{SDIST_SHA256}}"
  license "GPL-3.0-only"

  depends_on "uv" => :build
  depends_on "ffmpeg"
  depends_on "python@3.13"

  on_macos do
    depends_on arch: :arm64   # onnxruntime 1.30 has no Intel Mac wheels [V1]
    depends_on macos: :sonoma # onnxruntime's Apple Silicon wheels need macOS 14+ [A2]
  end

  resource "requirements" do
    url "https://raw.githubusercontent.com/rgoshen/lectr/v{{VERSION}}/requirements.lock"
    sha256 "{{LOCK_SHA256}}"
  end

  def install
    # Homebrew's Python helpers build every dependency from source, but onnxruntime is
    # published only as wheels, so install the pinned, hash-checked lock with uv instead.
    resource("requirements").stage { buildpath.install "requirements.lock" }
    python = Formula["python@3.13"].opt_bin/"python3.13"
    system "uv", "venv", libexec, "--python", python
    system "uv", "pip", "install", "--python", libexec/"bin/python",
           "--require-hashes", "--requirement", buildpath/"requirements.lock"
    system "uv", "pip", "install", "--python", libexec/"bin/python", "--no-deps", buildpath
    bin.install_symlink libexec/"bin/lectr"
  end

  test do
    assert_match "lectr #{version}", shell_output("#{bin}/lectr --version")
    assert_match "af_heart", shell_output("#{bin}/lectr voices")
  end
end
